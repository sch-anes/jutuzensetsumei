<script>
    let proceduresData = [];

    // Excelファイルの自動読み込み (procedures.xlsx)
    window.addEventListener('DOMContentLoaded', () => {
        fetch('procedures.xlsx')
            .then(response => {
                if (!response.ok) {
                    throw new Error('Excelファイルが見つかりません。');
                }
                return response.arrayBuffer();
            })
            .then(buffer => {
                const workbook = XLSX.read(buffer, { type: 'array' });
                const firstSheetName = workbook.SheetNames[0];
                const worksheet = workbook.Sheets[firstSheetName];
                
                // 1行目をヘッダーとしてJSON化
                proceduresData = XLSX.utils.sheet_to_json(worksheet);

                document.getElementById('status').textContent = `データ読込完了 (${proceduresData.length} 件登録済)`;
                const searchInput = document.getElementById('searchInput');
                searchInput.disabled = false;
                searchInput.focus();

                // 初期表示（全件表示または空）
                displayResults(proceduresData);
            })
            .catch(error => {
                console.error(error);
                document.getElementById('status').textContent = 'エラー: procedures.xlsx の読み込みに失敗しました。';
                document.getElementById('resultBody').innerHTML = '<tr><td colspan="2" class="no-result">Excelファイル（procedures.xlsx）を正しく配置してください。</td></tr>';
            });
    });

    // 検索入力時のイベント処理
    document.getElementById('searchInput').addEventListener('input', function(e) {
        const keyword = e.target.value.trim().toLowerCase();

        if (keyword === '') {
            displayResults(proceduresData);
            return;
        }

        // 術式名で部分一致検索（ひらがな/カタカナや英数小文字に対応）
        const filtered = proceduresData.filter(item => {
            const procedureName = String(item['術式名称'] || '').toLowerCase();
            return procedureName.includes(keyword);
        });

        displayResults(filtered);
    });

    // 結果テーブルの描画関数
    function displayResults(data) {
        const tbody = document.getElementById('resultBody');
        tbody.innerHTML = '';

        if (data.length === 0) {
            tbody.innerHTML = '<tr><td colspan="2" class="no-result">該当する術式は見つかりませんでした。</td></tr>';
            return;
        }

        data.forEach(item => {
            const tr = document.createElement('tr');
            
            const tdProcedure = document.createElement('td');
            tdProcedure.textContent = item['術式名称'] || '-';
            
            const tdAnesthesia = document.createElement('td');
            const span = document.createElement('span');
            span.className = 'tag';
            span.textContent = item['麻酔方法'] || '-';
            tdAnesthesia.appendChild(span);

            tr.appendChild(tdProcedure);
            tr.appendChild(tdAnesthesia);
            tbody.appendChild(tr);
        });
    }
</script>
